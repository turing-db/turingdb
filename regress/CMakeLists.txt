macro(regress_test REGRESS_NAME SOURCES)
    set( _SOURCES ${SOURCES} ${ARGN} ) 
    install(FILES ${_SOURCES}
        DESTINATION regress/${REGRESS_NAME})

    set(REGRESS_RUNSH ${CMAKE_CURRENT_SOURCE_DIR}/run.sh)

    # check that run.sh exists
    if (NOT EXISTS "${REGRESS_RUNSH}")
        message(FATAL_ERROR "Run script for the regress does not exist ${REGRESS_RUNSH}")
    endif()

    # add regress name to list
    set(REGRESS_RUNSH_INSTALL ${CMAKE_INSTALL_PREFIX}/regress/${REGRESS_NAME}/run.sh)
    list(APPEND REGRESS_LIST ${REGRESS_RUNSH_INSTALL})
    set(REGRESS_LIST ${REGRESS_LIST} PARENT_SCOPE)
endmacro()

# write run_regress target
# The script finds the wheel at runtime to avoid needing to re-run cmake after wheel build
set(RUN_REGRESS_CONTENT "#!/bin/bash\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}declare -a FAILED_TESTS=()\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}declare -i PASSED=0\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}export TURING_HOME=\"${CMAKE_INSTALL_PREFIX}\"\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}export PATH=\"\$TURING_HOME/bin:\$PATH\"\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}export SOURCE_DIR=\"${CMAKE_SOURCE_DIR}\"\n\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}# Find wheel at runtime (wheelhouse/ first, then dist/)\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}if [[ -z \"\$PYTURINGDB\" ]]\; then\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    PYTURINGDB=\$(ls \"\$SOURCE_DIR\"/wheel/*.whl 2>/dev/null | head -1)\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}fi\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}if [[ -z \"\$PYTURINGDB\" ]]\; then\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    PYTURINGDB=\$(ls \"\$SOURCE_DIR\"/dist/*.whl 2>/dev/null | head -1)\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}fi\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}if [[ -z \"\$PYTURINGDB\" || ! -f \"\$PYTURINGDB\" ]]\; then\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    echo \"Error: No wheel found in wheel/ or dist/. Build the wheel first with: uv build --wheel\" >&2\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    exit 1\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}fi\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}export PYTURINGDB\n\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}echo \"- Using turingdb: \$PYTURINGDB\"\n\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}# Each test runs once under the native (binary) client, except those listed\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}# in JSON_TESTS below which exercise the HTTP/JSON wire protocol directly.\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}# TURINGDB_TYPE is read by TuringDB() (json vs native); USE_TURING_PROTO tells the server which protocol to speak.\n\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}# Optional test filter: TURINGDB_TESTS=\"list_graphs call_metadata\" runs only those.\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}should_run() {\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    [[ -z \"\$TURINGDB_TESTS\" ]] && return 0\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    [[ \" \$TURINGDB_TESTS \" == *\" \$1 \"* ]]\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}}\n\n")
# Tests that exercise the HTTP/JSON wire protocol directly (e.g. by inspecting
# raw HTTP responses or hitting HTTP-only endpoints). They must run under the
# json client; all other tests run under native.
set(JSON_TESTS json_escape_properties get_nodes_invalid_id)

foreach(test IN ITEMS ${REGRESS_LIST})
    get_filename_component(test_dir "${test}" DIRECTORY)
    get_filename_component(test_name "${test_dir}" NAME)
    list(FIND JSON_TESTS "${test_name}" _json_idx)
    if(_json_idx GREATER -1)
        set(test_type "json")
        set(test_proto_flag "")
    else()
        set(test_type "native")
        set(test_proto_flag "1")
    endif()
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}if should_run ${test_name}\; then\n")
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    echo \"== type: ${test_type} :: ${test_name} ==\"\n")
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    if TURINGDB_TYPE=${test_type} USE_TURING_PROTO=${test_proto_flag} bash ${test}\; then\n")
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}        PASSED+=1\n")
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    else\n")
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}        FAILED_TESTS+=(\"[${test_type}] ${test_name}\")\n")
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}    fi\n")
    set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}fi\n")
endforeach()
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}\necho\necho \"=== SUMMARY ===\"\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}echo \"Passed: \$PASSED\"\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}echo \"Failed: \${#FAILED_TESTS[@]}\"\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}for t in \"\${FAILED_TESTS[@]}\"\; do echo \"  - \$t\"\; done\n")
set(RUN_REGRESS_CONTENT "${RUN_REGRESS_CONTENT}[[ \${#FAILED_TESTS[@]} -eq 0 ]]\n")

file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/run_regress.sh
           ${RUN_REGRESS_CONTENT})

install(FILES ${CMAKE_CURRENT_BINARY_DIR}/run_regress.sh
        DESTINATION regress)

add_custom_target(run_regress bash ${CMAKE_INSTALL_PREFIX}/regress/run_regress.sh
                  WORKING_DIRECTORY ${CMAKE_INSTALL_PREFIX}/regress)
